 end

\newcounter{question}

\newcount\dotcounter

\DeclareDocumentCommand{\newQuestion}{o m}{%
	\dotcounter1%
	\vskip 17pt \par\noindent%
	\textbf{\loop\ifnum\dotcounter < #2 · \ \ \advance\dotcounter by 1\repeat}%
	\refstepcounter{question}\ifnum#2>0{\hyperref[#1]{<}\ \ }\fi\Circled{\textbf{\thequestion}}\ %
}

\def\lastKey{}

\DeclareDocumentCommand{\identificationKey}{m m m m m m m m}{
	\newQuestion[#6]{#1}\ \ifdebug \ +#7\ -#8 \fi\ %
	\def\thisKey{#1}%
	\ifx\lastKey\thisKey\else\reversemarginpar\marginpar{\IfFileExists{keys/#2.pdf}{\vskip -4mm\includegraphics[width=30mm]{keys/#2.pdf}}{}}\fi%
	\ifnum#3=1 
		В знаке есть \expandafter\csname keyNomPl#2\endcsname?
	\else
		\newcount\adjustvalue
		\adjustvalue = #2
		\advance\adjustvalue by -1%
		Количество \expandafter\csname keyGenPl#2\endcsname\  в знаке больше \the\adjustvalue?
	\fi
	\par\noindent Да — \ref{#4}. Нет — \ref{#5}.%
	\global\def\lastKey{#2}%
}
